# Phase 4

- parse conditions

    example: if true { 1 } else { 0 } 
- parse loop
  - parse labels ('label)
  - break keyword
  - continue keyword

    example: loop { if i > 10 { break 10; } else { i += 1 } }
- ranges

# Phase 11

match <expr> {
  <pat> => <expr> // match arm pat
  <pat> if <expr> => <expr> // match arm guarded
  _ => <expr> // defualt arm
}

type MatchArm   = Box<Pat>
type MatchGuard = Option<Box<Expr>>
type MatchRhs   = Box<Expr>
ThinVec<(MatchArm, MatchGuard, MatchRhs)>
